// función reutilizable
function subirFoto($file, $carpeta) {
    if(isset($file) && $file['error'] == 0) {

        $nombre = $file['name'];
        $tmp = $file['tmp_name'];

        $ruta = $carpeta . uniqid() . "_" . $nombre;

        if(move_uploaded_file($tmp, $ruta)) {
            return $ruta;
        }
    }
    return null;
}
